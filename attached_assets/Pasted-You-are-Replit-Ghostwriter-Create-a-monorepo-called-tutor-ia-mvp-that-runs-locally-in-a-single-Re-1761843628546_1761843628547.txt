packages/providers/src/openai.ts
import OpenAI from "openai";
import { LLMProvider, ChatMessage, assertEnv } from "./index.js";
export function openaiProvider(): LLMProvider {
  const client = new OpenAI({ apiKey: assertEnv("OPENAI_API_KEY") });
  return {
    async chat({system, messages}:{system?:string; messages:ChatMessage[]}) {
      const msgs = [
        system ? { role:"system", content: system } : null,
        ...messages
      ].filter(Boolean) as any[];
      const r = await client.chat.completions.create({
        model: process.env.OPENAI_MODEL || "gpt-4o-mini",
        messages: msgs
      });
      return r.choices[0]?.message?.content ?? "";
    }
  };
}

packages/providers/src/gemini.ts
import { LLMProvider, ChatMessage, assertEnv } from "./index.js";
import { GoogleGenerativeAI } from "@google/generative-ai";
export function geminiProvider(): LLMProvider {
  const genAI = new GoogleGenerativeAI(assertEnv("GOOGLE_API_KEY"));
  const model = genAI.getGenerativeModel({ model: process.env.GEMINI_MODEL || "gemini-1.5-pro" });
  return {
    async chat({system, messages}:{system?:string; messages:ChatMessage[]}) {
      const content = (system?`[SYSTEM]\n${system}\n\n`:"") + messages.map(m=>`[${m.role.toUpperCase()}] ${m.content}`).join("\n");
      const r = await model.generateContent(content);
      return r.response.text();
    }
  };
}

packages/providers/src/factory.ts
import { openaiProvider } from "./openai.js";
import { geminiProvider } from "./gemini.js";
import type { LLMProvider } from "./index.js";
export function providerFor(name?:string): LLMProvider {
  const key = (name || process.env.PROVIDER_DEFAULT || "openai").toLowerCase();
  if (key.startsWith("gemini")) return geminiProvider();
  return openaiProvider();
}

apps/api/src/index.ts
import "dotenv/config";
import express from "express";
import cors from "cors";
import { providerFor } from "@tutor/providers/src/factory.js";
import { z } from "zod";

const app = express();
app.use(cors());
app.use(express.json());

app.get("/healthz", (_req,res)=>res.json({ok:true, provider: process.env.PROVIDER_DEFAULT || "openai"}));

const ChatSchema = z.object({
  messages: z.array(z.object({ role: z.enum(["system","user","assistant"]), content: z.string() })),
  provider: z.string().optional(),
  system: z.string().optional()
});

app.post("/api/tutor", async (req,res)=>{
  const parsed = ChatSchema.safeParse(req.body);
  if(!parsed.success) return res.status(400).json({error: parsed.error.format()});
  const { messages, provider, system } = parsed.data;
  try{
    const llm = providerFor(provider);
    const reply = await llm.chat({ system, messages });
    res.json({ reply });
  }catch(e:any){
    res.status(500).json({ error: e.message || "LLM error" });
  }
});

const GradeSchema = z.object({
  lessonId: z.string(),
  itemType: z.enum(["quiz"]),
  answerIndex: z.number(),
  userIndex: z.number()
});

import { scoreQuiz } from "@tutor/core/src/rubrics.js";
app.post("/api/grade", (req,res)=>{
  const p = GradeSchema.safeParse(req.body);
  if(!p.success) return res.status(400).json({error:p.error.format()});
  const { itemType, answerIndex, userIndex } = p.data;
  let score = 0;
  if(itemType==="quiz") score = scoreQuiz(answerIndex, userIndex);
  res.json({ score });
});

app.listen(3000, ()=>console.log("API listening on :3000"));

import React, { useState } from "react";
import { Renderer } from "../widgets/Renderer";
import demo from "../samples/lesson_demo.json";

export function App(){
  const [provider,setProvider] = useState<"openai"|"gemini">("openai");
  const [history,setHistory] = useState<{role:"user"|"assistant";content:string}[]>([]);
  async function ask(text:string){
    const res = await fetch("/api/tutor",{ method:"POST", headers:{"Content-Type":"application/json"}, body: JSON.stringify({ provider, system: SYSTEM_PROMPT, messages:[...history,{role:"user",content:text}] }) });
    const data = await res.json();
    setHistory(h=>[...h,{role:"assistant",content:data.reply}]);
  }
  return (
    <div className="max-w-3xl mx-auto p-6 space-y-4">
      <h1 className="text-2xl font-bold">Tutor IA MVP</h1>
      <div className="flex gap-2">
        <label className="text-sm">Proveedor:</label>
        <select className="border rounded px-2 py-1" value={provider} onChange={e=>setProvider(e.target.value as any)}>
          <option value="openai">OpenAI</option>
          <option value="gemini">Gemini</option>
        </select>
      </div>
      <Renderer dsl={demo}/>
      <div className="border rounded-xl p-3 space-y-2">
        <div className="font-semibold">Chat con Profe DANA</div>
        <input className="w-full border rounded px-3 py-2" placeholder="Escribe aquí…" onKeyDown={(e)=>{ if(e.key==="Enter"){ ask((e.target as HTMLInputElement).value); (e.target as HTMLInputElement).value=""; }}}/>
        <ul className="text-sm space-y-1">{history.map((m,i)=> <li key={i}><b>{m.role}:</b> {m.content}</li>)}</ul>
      </div>
    </div>
  );
}
const SYSTEM_PROMPT = `Eres “Profe DANA”, un tutor infantil. Explica breve y pregunta: «¿Quieres intentarlo conmigo?» Evita temas no aptos. No des respuestas de evaluaciones.`;

apps/web/src/widgets/Renderer.tsx
import React from "react";
import type { LessonDSL } from "@tutor/core/src/types";
import { Quiz } from "./quiz/Quiz";
import { OrderSteps } from "./quiz/OrderSteps";

export function Renderer({ dsl }: { dsl: LessonDSL }){
  return (
    <div className="space-y-4">
      {dsl.timeline.map((step, i) => {
        if (step.type === "tutor_say") return <div key={i} className="bg-white rounded-2xl p-4 shadow">🧑‍🏫 {step.text}</div>;
        if (step.type === "show_image") return <img key={i} src={step.src} alt={step.alt||""} className="rounded-2xl shadow"/>;
        if (step.type === "quiz") return <Quiz key={i} {...step} />;
        if (step.type === "interactive" && step.widget==="order-steps") return <OrderSteps key={i} {...step} />;
        if (step.type === "reflection") return <div key={i} className="italic">💭 {step.prompt}</div>;
        return null;
      })}
    </div>
  );
}

apps/web/src/widgets/quiz/Quiz.tsx
import React, { useState } from "react";
export function Quiz({ question, choices, answer }:{question:string; choices:string[]; answer:number}){
  const [picked,setPicked] = useState<number|null>(null);
  const [result,setResult] = useState<null|boolean>(null);
  async function submit(){
    const res = await fetch("/api/grade",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({itemType:"quiz", lessonId:"demo-01", answerIndex:answer, userIndex:picked})});
    const data = await res.json();
    setResult(data.score===1);
  }
  return (
    <div className="bg-white rounded-2xl p-4 shadow">
      <div className="font-semibold mb-2">❓ {question}</div>
      <div className="space-y-2">
        {choices.map((c,i)=>
          <label key={i} className={`block p-2 border rounded ${picked===i?"bg-gray-100":""}`}>
            <input type="radio" name="q" className="mr-2" onChange={()=>setPicked(i)} /> {c}
          </label>)}
      </div>
      <button className="mt-3 px-3 py-2 rounded bg-black text-white disabled:opacity-50" disabled={picked===null} onClick={submit}>Comprobar</button>
      {result!=null && <div className="mt-2">{result? "✅ ¡Muy bien!":"❌ Sigue intentando."}</div>}
    </div>
  );
}

apps/web/src/widgets/quiz/OrderSteps.tsx
import React, { useState } from "react";
export function OrderSteps({ data }:{ data:{steps:string[]; answer:number[]} }){
  const [order,setOrder] = useState<number[]>(data.steps.map((_,i)=>i));
  function move(i:number, dir:-1|1){
    const j = i+dir; if(j<0||j>=order.length) return;
    const next = [...order]; [next[i], next[j]] = [next[j], next[i]];
    setOrder(next);
  }
  const correct = JSON.stringify(order)===JSON.stringify(data.answer);
  return (
    <div className="bg-white rounded-2xl p-4 shadow">
      <div className="font-semibold mb-2">📐 Ordena los pasos</div>
      <ul className="space-y-1">
        {order.map((idx,i)=>(
          <li key={i} className="flex items-center gap-2">
            <span className="flex-1 border rounded px-2 py-1">{data.steps[idx]}</span>
            <div className="flex gap-1">
              <button className="px-2 py-1 border rounded" onClick={()=>move(i,-1)}>↑</button>
              <button className="px-2 py-1 border rounded" onClick={()=>move(i,1)}>↓</button>
            </div>
          </li>
        ))}
      </ul>
      <div className="mt-2">{correct? "✅ ¡Correcto!":"…"}</div>
    </div>
  );
}
